'''convertir une lettre en majuscule en utilisant son code ASCII'''

lettre = input("Entrer une lettre minuscule: ")
# code ascii de la lettre:
code = ord(lettre)
#on enleve 32 pour obtenir le code ascii de la majuscule correspondante
code -= 32
#on affiche la lettre en utilisant son code Ascii
print( chr(code))


'''convertir un mot en majuscules'''
mot = input("Entre un mot en lettres minuscules ")
MOT = ""
for l in mot:
    MOT = MOT + chr(ord(l)-32)
print(MOT)
